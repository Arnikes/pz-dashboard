"""Temporary verification of the accepted visual-style audit fixes."""

import json
from pathlib import Path

import pytest

from test_editors import editing, env, navigate  # noqa: F401

pytestmark = pytest.mark.browser
OUT = Path(__file__).resolve().parents[2] / ".tmp-visual-style-fix-evidence"


def styles(page, selector):
    return page.locator(selector).evaluate("""e => {
        const s=getComputedStyle(e), b=e.getBoundingClientRect();
        return {font:s.fontFamily,size:s.fontSize,weight:s.fontWeight,color:s.color,
            background:s.backgroundColor,border:s.borderColor,radius:s.borderRadius,
            outline:s.outlineStyle,outlineWidth:s.outlineWidth,height:b.height,
            width:b.width,top:b.top,left:b.left};
    }""")


def save(name, data):
    OUT.mkdir(exist_ok=True)
    (OUT / name).write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def settled(page):
    page.evaluate(
        "async()=>{await document.fonts.ready;document.getAnimations().forEach(a=>a.finish())}"
    )


@pytest.mark.parametrize("width", [390, 1440])
def test_danger_semantics(page, dashboard, width):
    page.set_viewport_size({"width": width, "height": 844})
    page.goto(dashboard["url"] + "/#/players")
    expect(page.locator("#btnStop")).to_be_enabled()
    page.evaluate("""() => {
        liveSource?.close();liveSource=null;clearTimeout(sseStartupTimer);
        renderPlayers({ok:true,names:['AuditPlayer'],count:1});
    }""")
    page.mouse.move(0, 0)
    ban = styles(page, '[data-p="ban"]')
    kick = styles(page, '[data-p="kick"]')
    assert ban["color"] == "rgb(224, 123, 110)"
    assert kick["color"] == "rgb(179, 167, 140)"
    page.locator('[data-p="ban"]').focus()
    focus = styles(page, '[data-p="ban"]')
    assert focus["outline"] == "solid" and focus["outlineWidth"] == "2px"
    page.keyboard.press("Enter")
    expect(page.locator("#modalOk")).to_have_class("btn solid-danger")
    expect(page.locator("#modalCancel")).to_be_focused()
    page.keyboard.press("Escape")
    expect(page.locator('[data-p="ban"]')).to_be_focused()
    navigate(page, "backups", width <= 740)
    page.evaluate("""renderBackups({ok:true,items:[{
        name:'audit-world.tar.gz',size:1024,mtime:'2026-10-06T00:00:00Z'}]})""")
    page.mouse.move(0, 0)
    delete = styles(page, '[data-b="del"]')
    assert delete["color"] == ban["color"]
    page.locator('[data-b="del"]').click()
    expect(page.locator("#modalOk")).to_have_class("btn solid-danger")
    page.keyboard.press("Escape")
    navigate(page, "console", width <= 740)
    prompt = styles(page, ".console-prompt")
    assert prompt["color"] == kick["color"]
    assert dashboard["actions"] == []
    save(
        f"step1-{width}.json",
        {"ban": ban, "kick": kick, "focus": focus, "delete": delete, "prompt": prompt},
    )

    data, _ = editing

    map_code = styles(page, "#modMapEditor code >> nth=0")

    page.locator("#mapList").fill("AuditTown;Muldraugh, KY")
    page.locator('#mapEdit button[type="submit"]').click()
    expect(page.locator("#draftSaved")).to_have_text(
        "Черновик сохранён" if language == "ru" else "Draft saved"

    assert not page.evaluate("document.documentElement.scrollWidth > innerWidth")
    assert (data / "Server/world.ini").read_bytes() == original
    assert dashboard["actions"] == []
    save(
        f"step2-{language}-{width}.json",
        {"maps": maps, "code": code, "map_code": map_code, "button": button},
    )


@pytest.mark.parametrize("width", [320, 390, 1181, 1440, 2048])
@pytest.mark.parametrize("language", ["ru", "en"])
def test_navigation_search_console(page, dashboard, editing, width, language):  # noqa: F811
    frame = {"ok": True, "text": "INFO ready\nWARN retry"}
    page.route("**/api/logs**", lambda route: route.fulfill(json=frame))
    page.add_init_script(f"localStorage.setItem('pz-language','{language}')")
    page.set_viewport_size({"width": width, "height": 844})
    page.goto(dashboard["url"] + "/#/mods")
    expect(page.locator("#configProfile")).to_have_value("world.ini")
    page.evaluate("liveSource?.close();liveSource=null;clearTimeout(sseStartupTimer)")
    metrics = {}
    if width <= 740:
        more = styles(page, "#navMore")
        link = styles(page, '.nav [data-route="mods"]')
        assert more["height"] == link["height"] == 48
        assert more["size"] == link["size"] == "10px"
        assert styles(page, "#navMore svg")["width"] == 19
        page.locator("#navMore").focus()
        page.keyboard.press("Enter")
        expect(page.locator("#navMore")).to_have_attribute("aria-expanded", "true")
        page.keyboard.press("Escape")
        expect(page.locator("#navMore")).to_be_focused()
        metrics["more"] = more
    page.locator('#modTabs [data-tab="order"]').click()
    toolbar = page.locator(".order-toolbar")
    assert toolbar.evaluate("e=>getComputedStyle(e).borderStyle") == "none"
    assert toolbar.evaluate("e=>getComputedStyle(e).position") == "sticky"
    search = toolbar.locator("input")
    search.fill("library")
    expect(page.locator("#modOrder .order-row")).to_have_count(1)
    search.fill("")
    metrics["toolbar"] = styles(page, ".order-toolbar")
    if width in (390, 1440):
        OUT.mkdir(exist_ok=True)
        page.evaluate(
            "async()=>{await document.fonts.ready;document.getAnimations().forEach(a=>a.finish())}"
        )
        page.screenshot(path=str(OUT / f"order-{language}-{width}.png"), full_page=True)
    navigate(page, "console", width <= 740)
    page.evaluate("renderLogs({ok:true,text:'INFO ready\\nWARN retry'})")
    page.evaluate(
        "async()=>{await document.fonts.ready;document.getAnimations().forEach(a=>a.finish())}"
    )
    console = styles(page, "#consoleOut")
    logs = styles(page, "#logsOut")
    logs_document_top = logs["top"] + page.evaluate("scrollY")
    if width >= 1181:
        assert abs(console["top"] - logs["top"]) < 1
        assert abs(console["width"] - logs["width"]) < 1
        title = styles(page, "#sec-console .card-head")
        tools = styles(page, "#quickCmds")
        assert title["height"] < 30
        assert abs(tools["top"] - title["top"] - title["height"] - 14) < 1
        assert abs(console["top"] - tools["top"] - tools["height"] - 14) < 2
    page.locator("#logsFilter").fill("missing")
    expect(page.locator("#logsResetFilters")).to_be_visible()
    assert abs(styles(page, "#logsOut")["top"] + page.evaluate("scrollY") - logs_document_top) < 1
    page.locator("#logsResetFilters").click()
    expect(page.locator("#logsFilter")).to_be_focused()
    expect(page.locator("#logsOut")).to_contain_text("ready")
    assert not page.evaluate("document.documentElement.scrollWidth > innerWidth")
    assert dashboard["actions"] == []
    metrics.update(console=console, logs=logs, footer=styles(page, ".logs-footer"))
    save(f"step3-{language}-{width}.json", metrics)
    if width in (390, 1440):
        page.evaluate(
            "async()=>{await document.fonts.ready;document.getAnimations().forEach(a=>a.finish())}"
        )
        page.screenshot(path=str(OUT / f"console-{language}-{width}.png"), full_page=True)

@pytest.mark.browser_context_args(service_workers="allow")
@pytest.mark.parametrize("width", [320, 390, 1440])
@pytest.mark.parametrize("language", ["ru", "en"])
def test_offline_header(page, context, pwa_server, width, language):  # noqa: F811
    page.add_init_script(f"localStorage.setItem('pz-language','{language}')")
    page.set_viewport_size({"width": width, "height": 844})
    ready(page, pwa_server)
    login_language = styles(page, ".language-switch")
    assert login_language["size"] == "16px" and login_language["height"] == 44
    settled(page)
    OUT.mkdir(exist_ok=True)
    if width in (390, 1440):
        page.screenshot(path=str(OUT / f"login-{language}-{width}.png"))
    context.set_offline(True)
    page.goto(pwa_server["url"] + "/")
    expect(page.locator(".pwa-offline h1")).to_have_text(
        "Нет связи с пультом" if language == "ru" else "Console disconnected"
    )
    settled(page)
    header = styles(page, ".pwa-header")
    heading = styles(page, ".pwa-offline h1")
    switch = styles(page, ".language-switch")
    assert abs(heading["top"] - header["top"] - header["height"] - 24) < 1
    assert styles(page, ".pwa-brand")["width"] == 36
    assert switch["height"] == 44
    assert not page.evaluate("document.documentElement.scrollWidth > innerWidth")
    page.locator(".language-switch").focus()
    assert styles(page, ".language-switch")["outline"] == "solid"
    if width in (390, 1440):
        page.screenshot(path=str(OUT / f"offline-{language}-{width}.png"))
    page.locator(".language-switch").select_option("en" if language == "ru" else "ru")
    expect(page.locator("html")).to_have_attribute("lang", "en" if language == "ru" else "ru")
    assert page.locator('script[src="/static/app.js"]').count() == 0
    save(
        f"step4-offline-{language}-{width}.json",
        {
            "header": header,
            "heading": heading,
            "language": switch,
            "login_language": login_language,
        },
    )
    context.set_offline(False)
